cmake_minimum_required(VERSION 3.16)
project(plan_env)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ego_ros2 REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(OpenCV REQUIRED)
add_library(plan_env SHARED src/grid_map.cpp src/raycast.cpp)
target_include_directories(plan_env PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(plan_env rclcpp ego_ros2 sensor_msgs nav_msgs geometry_msgs visualization_msgs cv_bridge message_filters pcl_conversions)
target_link_libraries(plan_env Eigen3::Eigen)
target_include_directories(plan_env SYSTEM PUBLIC ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(plan_env ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
install(TARGETS plan_env EXPORT export_plan_env ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_plan_env HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 rclcpp ego_ros2 sensor_msgs nav_msgs geometry_msgs visualization_msgs cv_bridge message_filters pcl_conversions PCL OpenCV)
ament_package()
